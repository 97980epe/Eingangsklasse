const functions = [
  {
    id: "a",
    term: "f(x) = x2 − 1",
    domain: "D = [−2; 2]",
    range: "W = [−1; 3]",
    dmin: -2, dmax: 2, rmin: -1, rmax: 3,
    fn: x => x * x - 1
}, {
    id: "b",
    term: "g(x) = −x + 1",
    domain: "D = [−2; 2]",
    range: "W = [−1; 3]",
    dmin: -2, dmax: 2, rmin: -1, rmax: 3,
    fn: x => -x + 1
}, {
    id: "c",
    term: "h(x) = |x|",
    domain: "D = [−2; 2]",
    range: "W = [0; 2]",
    dmin: -2, dmax: 2, rmin: 0, rmax: 2,
    fn: x => Math.abs(x)
}, {
    id: "d",
    term: "p(x) = x3",
    domain: "D = [−1; 1]",
    range: "W = [−1; 1]",
    dmin: -1, dmax: 1, rmin: -1, rmax: 1,
    fn: x => x * x * x
} ];
const kinds = ["graph", "term", "domain", "range"];
const kindNames = {
  graph: "Funktionsgraph",
  term: "Funktionsterm",
  domain: "Definitionsmenge",
 
   range: "Wertemenge"
};
function graphSvg(item) {
  const width = 180, height = 92;
  const left = 18, right = 169, top = 8, bottom = 78;
  const xPad = (item.dmax - item.dmin) * 0.12;
  const ySpan = Math.max(item.rmax - item.rmin, 1);
  const yPad = ySpan * 0.18;
  const xmin = item.dmin - xPad, xmax = item.dmax + xPad;
  const ymin = item.rmin - yPad, ymax = item.rmax + yPad;
  const X = x => left + (x - xmin) / (xmax - xmin) * (right - left);
  const Y = y => bottom - (y - ymin) / (ymax - ymin) * (bottom - top);
  let path = "";
  const steps = 100;
  for (let i = 0; i <= steps; i++) {
    const x = item.dmin + (item.dmax - item.dmin) * i / steps;
    const y = item.fn(x);
    path += (i === 0 ? "M" : "L") + X(x).toFixed(2) + " " + Y(y).toFixed(2)
+ " "; }
  const xAxisY = ymin <= 0 && ymax >= 0 ? Y(0) : bottom;
  const yAxisX = xmin <= 0 && xmax >= 0 ? X(0) : left;
  return `<svg viewBox="0 0 ${width} ${height}" role="img" aria-label="Funk
tionsgraph">
    <line class="axis" x1="${left}" y1="${xAxisY}" x2="${right}" y2="${xAxi
sY}"/>
    <line class="axis" x1="${yAxisX}" y1="${top}" x2="${yAxisX}" y2="${bott
om}"/>
    <path class="curve" d="${path}"/>
    <circle class="endpoint" cx="${X(item.dmin)}" cy="${Y(item.fn(item.dmi
n))}" r="2.8"/>
    <circle class="endpoint" cx="${X(item.dmax)}" cy="${Y(item.fn(item.dma
x))}" r="2.8"/>
  </svg>`;
}
function makeCards() {
  return functions.flatMap(item =>
    kinds.map(kind => ({ id: item.id, kind, item }))
  ).sort(() => Math.random() - 0.5);
}
let cards = [];
let selected = [];
let matched = new Set();
let locked = false;
let attempts = 0;
 
 function cardMarkup(card, index) {
  let value = "";
  if (card.kind === "graph") value = `<div class="graph">${graphSvg(card.it
em)}</div>`;
  if (card.kind === "term") value = `<div class="value">${card.item.term}</
div>`;
  if (card.kind === "domain") value = `<div class="value">${card.item.domai
n}</div>`;
  if (card.kind === "range") value = `<div class="value">${card.item.range}
</div>`;
  const isMatched = matched.has(index);
  const isSelected = selected.includes(index);
  const stateClass = isMatched ? "correct" : isSelected ? "revealed" : "cov
ered";
  return `<button type="button" class="card ${stateClass}" data-index="${in
dex}"
      aria-label="${isMatched || isSelected ? kindNames[card.kind] + ": " +
(card.kind === "term" ? card.item.term : card.kind === "domain" ? card.ite
m.domain : card.kind === "range" ? card.item.range : "Graph") : "Verdeckte
Karte"}"
      ${isMatched ? "disabled" : ""}>
      <span class="content"><span class="kind">${kindNames[card.kind]}</spa
n>${value}</span>
    </button>`;
}
function render() {
  const grid = document.getElementById("grid");
  grid.innerHTML = cards.map(cardMarkup).join("");
  document.getElementById("score").textContent =
    `Gefundene Quartette: ${matched.size / 4} von 4`;
  grid.querySelectorAll(".card").forEach(button => {
    button.addEventListener("click", () => flip(Number(button.dataset.inde
x)));
}); }
function flip(index) {
  if (locked || matched.has(index) || selected.includes(index)) return;
  selected.push(index);
  render();
  if (selected.length < 4) {
    document.getElementById("message").textContent =
      `Noch ${4 - selected.length} Karte${selected.length === 3 ? "" : "n"}
auswählen.`;
    return;
}
 
    locked = true;
  attempts++;
  const chosen = selected.map(i => cards[i]);
  const sameFunction = chosen.every(card => card.id === chosen[0].id);
  const allKinds = new Set(chosen.map(card => card.kind)).size === 4;
  if (sameFunction && allKinds) {
    selected.forEach(i => matched.add(i));
    selected = [];
    locked = false;
    render();
    document.getElementById("message").textContent =
      matched.size === cards.length
        ? `Alle Quartette gefunden! Du hast ${attempts} Versuche gebraucht.
`
      : "Richtig zugeordnet – die Karten leuchten grün!";
} else {
  selected.forEach(i => {
    const button = document.querySelector(`[data-index="${i}"]`);
    if (button) button.classList.add("wrong");
  });
  document.getElementById("message").textContent =
    "Das passt noch nicht zusammen – die Karten leuchten rot.";
  setTimeout(() => {
    selected = [];
    locked = false;
    render();
    document.getElementById("message").textContent = "Wähle vier Karten a
us.";
    }, 1100);
} }
function startGame() {
  cards = makeCards();
  selected = [];
  matched = new Set();
  locked = false;
  attempts = 0;
  document.getElementById("message").textContent = "Wähle vier Karten au
s.";
  render();
}
document.getElementById("reset").addEventListener("click", startGame);
startGame();
